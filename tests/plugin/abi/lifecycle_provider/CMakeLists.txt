cmake_minimum_required(VERSION 3.22)
project(LifecycleProvider VERSION 1.0.0 LANGUAGES CXX)
find_package(RoDAQ REQUIRED CONFIG)
find_package(nlohmann_json 3.10.5 EXACT REQUIRED CONFIG)
add_library(lifecycle SHARED provider.cpp)
target_link_libraries(lifecycle PRIVATE nlohmann_json::nlohmann_json)
target_compile_options(lifecycle PRIVATE -Wall -Wextra -Wpedantic -Werror)
rodaq_register_plugin(lifecycle PLUGIN_ID com.example.lifecycle PLUGIN_NAME "Lifecycle fault fixture")
