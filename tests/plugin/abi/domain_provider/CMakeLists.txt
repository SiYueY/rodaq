cmake_minimum_required(VERSION 3.22)
project(DomainCompatibilityProvider VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(RoDAQ REQUIRED CONFIG)
add_library(domain_provider SHARED provider.c)
target_compile_definitions(domain_provider PRIVATE DOMAIN_STORAGE=${DOMAIN_STORAGE} FIXTURE_MINOR=${FIXTURE_MINOR})
target_compile_options(domain_provider PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
if(DOMAIN_STORAGE)
    set(domain_id com.example.storage.compat)
else()
    set(domain_id com.example.acquisition.compat)
endif()
rodaq_register_plugin(domain_provider PLUGIN_ID ${domain_id} PLUGIN_NAME "Domain compatibility fixture")
