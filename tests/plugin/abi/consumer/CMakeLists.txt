cmake_minimum_required(VERSION 3.22)
project(CounterConsumer LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(RoDAQ REQUIRED CONFIG)
add_executable(consumer consumer.c)
target_include_directories(consumer PRIVATE $<TARGET_PROPERTY:rodaq::plugin,INTERFACE_INCLUDE_DIRECTORIES>)
target_compile_definitions(consumer PRIVATE FIXTURE_MINOR=${FIXTURE_MINOR})
target_compile_options(consumer PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
target_link_libraries(consumer PRIVATE ${CMAKE_DL_LIBS})
