cmake_minimum_required(VERSION 3.22)
project(CounterProvider VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(RoDAQ REQUIRED CONFIG)
add_library(counter SHARED provider.c)
target_compile_definitions(counter PRIVATE FIXTURE_MINOR=${FIXTURE_MINOR})
target_compile_options(counter PRIVATE -Wall -Wextra -Wpedantic -Werror)
rodaq_register_plugin(counter PLUGIN_ID com.example.counter PLUGIN_NAME "Counter fixture")
