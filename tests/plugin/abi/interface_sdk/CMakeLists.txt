cmake_minimum_required(VERSION 3.22)
project(CounterInterfaceSdk VERSION 1.0.0 LANGUAGES CXX)
find_package(RoDAQ REQUIRED CONFIG)
add_library(counter_interface INTERFACE)
add_library(Counter::interface ALIAS counter_interface)
target_link_libraries(counter_interface INTERFACE rodaq::plugin)
target_include_directories(counter_interface INTERFACE $<INSTALL_INTERFACE:include/counter>)
set_target_properties(counter_interface PROPERTIES EXPORT_NAME interface)
install(FILES counter.h counter.hpp DESTINATION include/counter)
install(TARGETS counter_interface EXPORT CounterTargets)
install(EXPORT CounterTargets NAMESPACE Counter:: DESTINATION lib/cmake/CounterInterface)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/CounterInterfaceConfig.cmake
    "include(CMakeFindDependencyMacro)\nfind_dependency(RoDAQ CONFIG)\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/CounterTargets.cmake\")\n")
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/CounterInterfaceConfig.cmake DESTINATION lib/cmake/CounterInterface)
