cmake_minimum_required(VERSION 3.22)
project(DomainCompatibilityConsumer LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(RoDAQ REQUIRED CONFIG)
add_executable(domain_consumer consumer.c)
target_include_directories(domain_consumer PRIVATE $<TARGET_PROPERTY:rodaq::plugin,INTERFACE_INCLUDE_DIRECTORIES>)
target_compile_definitions(domain_consumer PRIVATE DOMAIN_STORAGE=${DOMAIN_STORAGE} FIXTURE_MINOR=${FIXTURE_MINOR})
target_compile_options(domain_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
target_link_libraries(domain_consumer PRIVATE ${CMAKE_DL_LIBS})
