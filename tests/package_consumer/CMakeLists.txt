cmake_minimum_required(VERSION 3.22)
project(RoDAQPackageConsumer LANGUAGES CXX)
if(RODAQ_COMPONENT_MODE STREQUAL "required")
    find_package(RoDAQ CONFIG REQUIRED COMPONENTS ros2_support)
elseif(RODAQ_COMPONENT_MODE STREQUAL "optional")
    find_package(RoDAQ CONFIG REQUIRED OPTIONAL_COMPONENTS ros2_support)
elseif(RODAQ_COMPONENT_MODE STREQUAL "unknown")
    find_package(RoDAQ CONFIG REQUIRED COMPONENTS not_a_component)
else()
    find_package(RoDAQ CONFIG REQUIRED)
endif()
get_property(imported_targets DIRECTORY PROPERTY IMPORTED_TARGETS)
set(rodaq_targets "")
foreach(target IN LISTS imported_targets)
    if(target MATCHES "^rodaq::")
        list(APPEND rodaq_targets "${target}")
    endif()
endforeach()
list(SORT rodaq_targets)
set(expected_targets "rodaq::daq;rodaq::plugin")
if(RODAQ_EXPECT_ROS2)
    list(APPEND expected_targets rodaq::ros2_support)
    if(NOT RoDAQ_ros2_support_FOUND OR NOT TARGET ament_index_cpp::ament_index_cpp)
        message(FATAL_ERROR "Requested ROS 2 component did not resolve")
    endif()
elseif(TARGET ament_index_cpp::ament_index_cpp OR TARGET rodaq::ros2_support)
    message(FATAL_ERROR "Unrequested or unavailable ROS 2 component leaked into base SDK")
endif()
if(NOT rodaq_targets STREQUAL expected_targets)
    message(FATAL_ERROR "Unexpected public targets: ${rodaq_targets}; expected ${expected_targets}")
endif()
add_executable(consumer main.cpp)
if(RODAQ_SANITIZER STREQUAL "address-undefined")
    target_compile_options(consumer PRIVATE -fsanitize=address,undefined)
    target_link_options(consumer PRIVATE -fsanitize=address,undefined)
elseif(RODAQ_SANITIZER STREQUAL "thread")
    target_compile_options(consumer PRIVATE -fsanitize=thread)
    target_link_options(consumer PRIVATE -fsanitize=thread)
endif()
target_link_libraries(consumer PRIVATE rodaq::daq rodaq::plugin)
set_target_properties(consumer PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)

if(RODAQ_EXPECT_ROS2)
    add_executable(ros2_consumer ros2.cpp)
    target_link_libraries(ros2_consumer PRIVATE rodaq::ros2_support)
    if(RODAQ_SANITIZER STREQUAL "address-undefined")
        target_compile_options(ros2_consumer PRIVATE -fsanitize=address,undefined)
        target_link_options(ros2_consumer PRIVATE -fsanitize=address,undefined)
    elseif(RODAQ_SANITIZER STREQUAL "thread")
        target_compile_options(ros2_consumer PRIVATE -fsanitize=thread)
        target_link_options(ros2_consumer PRIVATE -fsanitize=thread)
    endif()
endif()
